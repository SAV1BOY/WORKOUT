function juntar(vs) {
  const ok = vs.filter(Boolean)
  return {
    aprovado: ok.length === 2 && ok.every(v => v.aprovado),
    bloqueantes: ok.flatMap(v => v.bloqueantes.map(x => `[${v.lente}] ${x}`)),
    importantes: ok.flatMap(v => v.importantes.map(x => `[${v.lente}] ${x}`)),
    menores: ok.flatMap(v => v.menores.map(x => `[${v.lente}] ${x}`)),
    faltou_lente: ok.length < 2,
  }
}
